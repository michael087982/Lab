// ===== Fill these in (see SETUP.md) =====
const CONFIG = {
  LIFF_ID: '2012198089-geTm3ZEy',
  API_URL: 'https://script.google.com/macros/s/AKfycbwzQLQe3Q6Cq4xD5XkUuHvwC1PXaW7j5Mq_-gkyHSY2qYL9nXpK2Hsw3-rxhZctVqyI/exec',     // ends with /exec
  OA_ADD_FRIEND_URL: 'https://line.me/R/ti/p/@338xlpvl',
  TIMEZONE: 'Asia/Bangkok',
};

const DEMO = new URLSearchParams(location.search).has('demo');  // open with ?demo to preview without LINE
const $ = id => document.getElementById(id);
let state = null;
let notFriend = false;

init();

async function init() {
  try {
    if (!DEMO) {
      await liff.init({ liffId: CONFIG.LIFF_ID });
      if (!liff.isLoggedIn()) { liff.login({ redirectUri: location.href }); return; }
    }
    const s = await api('session');
    sessionStorage.removeItem('relogin');
    s.linked ? showChecklist(s) : showLink(s.lineName);
    if (!DEMO) checkFriendship();
  } catch (e) {
    show('fatal');
    $('fatalMsg').textContent = e.message || String(e);
  }
}

async function api(action, data = {}) {
  if (DEMO) return demoApi(action, data);
  const res = await fetch(CONFIG.API_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },  // avoids a CORS preflight Apps Script can't answer
    body: JSON.stringify(Object.assign({ action, idToken: liff.getIDToken() }, data)),
  });
  const json = await res.json();
  if (!json.ok) {
    if (json.error === 'auth' && !sessionStorage.getItem('relogin')) {
      sessionStorage.setItem('relogin', '1');
      liff.logout();
      location.reload();
      return new Promise(() => {});
    }
    throw new Error(json.message);
  }
  return json;
}

async function checkFriendship() {
  try {
    const f = await liff.getFriendship();
    notFriend = !f.friendFlag;
    if (state) renderBanners();
  } catch (_) { /* friendship check needs the OA linked to the LINE Login channel */ }
}

function show(id) {
  ['loading', 'fatal', 'link', 'app'].forEach(s => { $(s).hidden = s !== id; });
}

// ----- Link screen -----
function showLink(lineName) {
  show('link');
  $('hello').textContent = lineName ? `Welcome, ${lineName}` : 'Welcome';
  $('email').focus();
}

$('linkForm').addEventListener('submit', async e => {
  e.preventDefault();
  const email = $('email').value.trim();
  $('linkErr').textContent = '';
  if (!/^\S+@\S+\.\S+$/.test(email)) { $('linkErr').textContent = 'Please enter a valid email.'; return; }
  $('linkBtn').disabled = true;
  try {
    showChecklist(await api('link', { email }));
  } catch (err) {
    $('linkErr').textContent = err.message;
  } finally {
    $('linkBtn').disabled = false;
  }
});

// ----- Checklist -----
function showChecklist(s) {
  state = s;
  show('app');
  $('who').textContent = s.staff.name || s.staff.email;
  $('when').textContent = formatDate(s.date) + ' Â· shift ends ' + s.shiftEnd;

  const done = s.tasks.filter(t => t.checkedAt).length;
  $('doneCount').textContent = done;
  $('totalCount').textContent = s.tasks.length;
  $('barFill').style.width = (s.tasks.length ? done / s.tasks.length * 100 : 0) + '%';
  $('bar').classList.toggle('full', done === s.tasks.length && done > 0);

  const ul = $('tasks');
  ul.replaceChildren(...s.tasks.map(taskRow));
  renderBanners();
}

function taskRow(t) {
  const li = document.createElement('li');
  li.innerHTML = `
    <label class="task">
      <input type="checkbox">
      <span class="box"><svg viewBox="0 0 16 16" fill="none" stroke="#fff" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M3 8.5l3.2 3L13 4.5"/></svg></span>
      <span class="body"><span class="title"></span><span class="details"></span></span>
      <span class="time"></span>
    </label>`;
  const input = li.querySelector('input');
  input.checked = !!t.checkedAt;
  li.querySelector('.title').textContent = t.task;
  li.querySelector('.details').textContent = t.details;
  li.querySelector('.time').textContent = t.checkedAt ? 'â ' + t.checkedAt : '';
  input.addEventListener('change', () => toggle(t, input));
  return li;
}

async function toggle(t, input) {
  input.disabled = true;
  try {
    showChecklist(await api('check', { taskId: t.id, checked: input.checked }));
  } catch (err) {
    input.checked = !input.checked;
    input.disabled = false;
    toast(err.message);
  }
}

function renderBanners() {
  const box = $('banners');
  box.replaceChildren();
  const left = state.tasks.filter(t => !t.checkedAt).length;

  if (notFriend) {
    const b = banner('info', 'Add our LINE Official Account to get shift-end reminders.');
    const btn = document.createElement('button');
    btn.textContent = 'Add friend';
    btn.onclick = () => liff.openWindow({ url: CONFIG.OA_ADD_FRIEND_URL, external: false });
    b.append(btn);
    box.append(b);
  }
  if (state.tasks.length && left === 0) {
    box.append(banner('good', 'All tasks done for today. Thank you!'));
  } else if (nowHHMM() >= state.shiftEnd) {
    box.append(banner('warn', `Shift has ended. ${left} task${left > 1 ? 's' : ''} still unchecked.`));
  }
}

function banner(kind, text) {
  const d = document.createElement('div');
  d.className = 'banner ' + kind;
  const span = document.createElement('span');
  span.textContent = text;
  d.append(span);
  return d;
}

function nowHHMM() {
  return new Intl.DateTimeFormat('en-GB', { timeZone: CONFIG.TIMEZONE, hour: '2-digit', minute: '2-digit', hourCycle: 'h23' })
    .format(new Date());
}

function formatDate(ymd) {
  return new Date(ymd + 'T00:00:00Z').toLocaleDateString('en-GB',
    { weekday: 'long', day: 'numeric', month: 'short', year: 'numeric', timeZone: 'UTC' });
}

let toastTimer;
function toast(msg) {
  $('toast').textContent = msg;
  $('toast').hidden = false;
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { $('toast').hidden = true; }, 3500);
}

async function demoApi(action, data) {
  await new Promise(r => setTimeout(r, 250));
  const snapshot = () => ({
    ok: true, linked: true, staff: { email: demoData.email, name: 'Demo Staff' },
    date: new Date().toLocaleDateString('en-CA', { timeZone: CONFIG.TIMEZONE }), shiftEnd: '17:00',
    tasks: demoData.tasks.map(t => Object.assign({}, t)),
  });
  if (action === 'session') return demoData.linked ? snapshot() : { ok: true, linked: false, lineName: 'Demo User' };
  if (action === 'link') { demoData.linked = true; demoData.email = data.email; return snapshot(); }
  if (action === 'check') {
    demoData.tasks.find(t => t.id === data.taskId).checkedAt = data.checked ? nowHHMM() : null;
    return snapshot();
  }
}
